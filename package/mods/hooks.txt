# Translation-time hooks (docs/hooks.md): one `ADDRESS name  # note` per line. An address is the entry of a guest function (the
# translated function calls the hook on entry; the callback can read and change the registers and the guest's memory).
# `in:ADDRESS name` hooks the function containing that address. Re-run `mh3u-tools translate` after changing this list (the launcher
# does it when the list changes). These are the hooks the launcher's Cheats and Mods sections use (mh3u-rt/src/cheats.rs, combat.rs).

# Zenny (the "more zenny" cheat). r3 is the save, r4 the signed change, r5 = &money.
0x0219c720 money_add          # r4 = signed change in zenny (-200 craft fee, -50 meal, -310 quest fee, +310 refund), r5 = &money

# Items (the "infinite items" cheat): the function that puts an item in the pouch or takes it out; a count of -1 is an item used.
in:0x0219b080 item_to_pouch      # f_0219aeac(save, item, count): count -1 = an item used (run 39, twice, from lr 0x285edac, f_0285ecd4)

# Combat (the life bars, the damage meter and the one hit kill / infinite sharpness cheats).
0x02129ef4 hit_resolve           # f_02129ef4: resolves ONE hit; r5 = the hit record: byte +6 = attacker kind (0 hunter, 1 monster), float +0x44 = base damage
0x0259fc98 monster_init           # f_0259fc98(monster, ..): sets a monster's life at spawn (uEnemy::initMaxHp); r3 = the monster, whose state *(r3+0x1be0) holds hp_vital (+0x8b8) and max_hp_vital (+0x8bc). The life bars and the damage meter follow the monsters from here
0x0285f530 sharp_change          # f_0285f530(hunter, change, flag): adds `change` to the weapon's sharpness (negative = it wears)

# Bowgun aim (always on): f_028a19b8(hunter) is the per-frame update of a ranged weapon's aim; the hook moves the target pitch from the pad.
0x028a19b8 aim_pitch          # r3 = hunter

# Camera (always on unless MH3U_ANALOG_CAMERA=off): the camera follows how far the right stick is pushed. Both are f(owner, camera, hunter).
0x02286f18 camera_turn        # r4 = camera, r5 = hunter: the turn (left/right) from the stick's orders
0x02286890 camera_pitch       # r4 = camera, r5 = hunter: up/down (six heights on foot, a free tilt in the water)

# Hold to gather (always on unless MH3U_HOLD_GATHER=off): every item carved, gathered or mined goes through the cockpit's "obtained" step.
0x020eca2c obtain_item        # f_020eca2c(cockpit, _, item, ...): from an item on, a held A reaches the game as presses

# Quest map (the map switch in the game's menu, MH3U_MAP=on): whether the map is shown; the hook says yes without the Map item of the supply box.
0x0260eea8 map_available      # f_0260eea8(_, hunter): 1 = the map is shown (sGame+0x1b9 = 1, or the hunter has the Map item 0x3f2, never on stage kind 6)

# Screen filters (the bloom and blur switches in the game's menu; MH3U_BLOOM=off / MH3U_BLUR=off): the draw of each filter, which the hook skips when it is off.
0x02a7830c bloom_draw         # uBloomFilter::vf11: the glow around bright things
0x02a8504c dof_draw           # uDOFFilter::vf11: depth of field (the game uses it in few scenes)
0x02a79f90 bokeh_draw         # uBokehFilter::vf11: bokeh blur
0x0259fb84 monster_max          # f_0259fb84 = uEnemy::setMaxHp_(monster, base): r3 = the monster, r4 = its base life (max = base x the row of the quest). Only a probe for now
